# Етап 1: Побудова залежностей
FROM python:3.11-slim as builder

WORKDIR /app
RUN pip install --no-cache-dir uv

COPY pyproject.toml ./
RUN uv pip install --system --no-cache -e .

# Етап 2: Фінальний образ
FROM python:3.11-slim

WORKDIR /app

# Створення не-root користувача для безпеки
RUN useradd -m findexuser
USER findexuser

COPY --chown=findexuser:findexuser . .
RUN pip install --no-cache-dir . fastapi uvicorn pydantic jinja2 python-multipart

ENV INDEX_PATH=/app/index.bin
EXPOSE 8000

CMD ["python", "-m", "findex.cli", "serve", "--host", "0.0.0.0", "--port", "8000"]
